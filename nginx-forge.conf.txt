# Bloque para el nginx del sitio en Forge (Sitio → Files → Edit Nginx Configuration).
# Va en lugar del `location / { try_files $uri $uri/ =404; }` que trae un sitio «Other» de Forge: todo va a Evolution.
# El resto (ssl, headers, include de forge-conf, logs, el deny de archivos ocultos) queda como está.
# Si el sitio fuera PHP, además se borra el `location ~ \.php$ { ... }`.

client_max_body_size 64M;

# ⛔ La raíz NO se redirige al manager: el manager valida la «URL del Servidor» pidiendo esa raíz y esperando el JSON de
# Evolution; con un 302 contesta «Servidor inválido» y no deja entrar (2026-10-22, en producción). A la gente se le pasa
# directo el link a /manager/.
location / {
    proxy_pass http://127.0.0.1:8080;
    proxy_http_version 1.1;

    # WebSocket (eventos en vivo de Evolution).
    proxy_set_header Upgrade $http_upgrade;
    proxy_set_header Connection "upgrade";

    proxy_set_header Host $host;
    proxy_set_header X-Real-IP $remote_addr;
    proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
    proxy_set_header X-Forwarded-Proto $scheme;

    proxy_read_timeout 120s;
    proxy_send_timeout 120s;
}
